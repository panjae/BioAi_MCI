""" response = get_gpt_response(prompt) self.update_profile(response) return response def suggest_topics(self): # 개인화된 대화 주제 생성 topics = [] for interest in self.user_profile['interests']: topic_prompt = f"{interest}에 대한 흥미로운 대화 주제 3개" topics.extend(get_topic_suggestions(topic_prompt)) return topics

# Week 6-1: LT(사랑대화기) 모듈 class LoveTalker: def __init__(self): self.emotional_state = "neutral" self.conversation_depth = 0 def therapeutic_conversation(self, user_input, user_profile): # 7단계 감정 심화 대화 prompts_by_depth = { 1: "공감과 경청", 2: "감정 탐색", 3: "경험 회상", 4: "의미 발견", 5: "긍정 강화", 6: "통찰 유도", 7: "미래 희망" } current_approach = prompts_by_depth[min(self.conversation_depth + 1, 7)] response = self.generate_therapeutic_response( user_input, user_profile, current_approach ) self.conversation_depth += 1 return response

MS 모듈: 인지 평가 시스템

# Week 6-2: MS(인지측정기) 모듈 class MindScanner: def __init__(self): self.cognitive_scores = { 'memory': [], 'attention': [], 'language': [], 'executive': [] } def memory_test(self): # 단어 목록 기억 테스트 words = random.sample(["사과", "책상", "하늘", "음악", "행복", "여행", "친구", "커피", "산책", "바다"], 5) st.write("🧠 기억력 테스트") st.write(f"다음 단어들을 30초간 기억해주세요:") cols = st.columns(5) for i, word in enumerate(words): cols[i].info(word) time.sleep(30) st.empty() # 간섭 과제 st.write("잠시 다른 이야기를 해볼까요?") distractor = st.text_input("오늘 날씨는 어떤가요?") # 회상 테스트 recalled = st.text_input("아까 본 단어들을 모두 적어주세요:") score = self.calculate_memory_score(words, recalled) self.cognitive_scores['memory'].append(score) return score def attention_test(self): # 숫자 거꾸로 따라하기 (Digit Span) st.write("🎯 주의력 테스트") difficulty = len(self.cognitive_scores['attention']) + 3 numbers = [random.randint(1, 9) for _ in range(difficulty)] st.write(f"숫자를 거꾸로 말해주세요: {' '.join(map(str, numbers))}") answer = st.text_input("답:") correct = ' '.join(map(str, reversed(numbers))) score = 100 if answer.replace(' ', '') == correct.replace(' ', '') else 0 self.cognitive_scores['attention'].append(score) return score def generate_report(self): # 인지 기능 평가 리포트 생성 report = { 'overall_score': np.mean([ np.mean(scores) for scores in self.cognitive_scores.values() if scores ]), 'memory_trend': self.calculate_trend(self.cognitive_scores['memory']), 'attention_trend': self.calculate_trend(self.cognitive_scores['attention']), 'recommendation': self.get_recommendation() } return report

# Week 7: 3개 모듈 통합 class MFLRSystem: def __init__(self): self.hf = HeartFinder() # 마음발굴기 self.lt = LoveTalker() # 사랑대화기 self.ms = MindScanner() # 인지측정기 self.session_data = [] def run_session(self): st.title("🧠 MFLR 인지건강 도우미") # 탭 구성 tab1, tab2, tab3, tab4 = st.tabs(["💬 대화", "🧠 인지평가", "📊 리포트", "⚙️ 설정"]) with tab1: # HF + LT 통합 대화 user_input = st.text_input("오늘 하루는 어떠셨나요?") if user_input: # 1. 관심사 추출 (HF) self.hf.extract_interests(user_input) # 2. 치료적 대화 생성 (LT) response = self.lt.therapeutic_conversation( user_input, self.hf.user_profile ) st.write(f"AI: {response}") # 3. 대화 중 자연스럽게 인지 과제 삽입 if len(self.session_data) % 5 == 4: st.info("잠깐, 간단한 두뇌 운동을 해볼까요?") if st.button("시작하기"): self.ms.memory_test() with tab2: # MS 독립 실행 st.subheader("인지 기능 평가") col1, col2 = st.columns(2) with col1: if st.button("🧠 기억력 테스트"): score = self.ms.memory_test() st.metric("기억력 점수", f"{score}점") with col2: if st.button("🎯 주의력 테스트"): score = self.ms.attention_test() st.metric("주의력 점수", f"{score}점") with tab3: # 통합 리포트 st.subheader("나의 인지건강 리포트") if self.ms.cognitive_scores['memory']: report = self.ms.generate_report() # 점수 시각화 st.metric("종합 점수", f"{report['overall_score']:.1f}점") # 트렌드 차트 chart_data = pd.DataFrame({ '기억력': self.ms.cognitive_scores['memory'], '주의력': self.ms.cognitive_scores['attention'] }) st.line_chart(chart_data) # 추천사항 st.info(f"💡 {report['recommendation']}")
